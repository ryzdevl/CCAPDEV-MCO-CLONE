    <script>
        // Post modal link functionality
        document.getElementById('post-modal-link').addEventListener('click', function(e) {
            e.preventDefault();
            // This will work if the parent page has the post modal
            if (window.parent && window.parent.$) {
                window.parent.$("#post-iframe").attr("src", " PostModal.html");
                window.parent.$("#post-composer-modal").fadeIn(300);
            } else {
                // Fallback if not in iframe
                window.location.href = ' PostModal.html';
            }
        });
    </script>

    <script>
        const searchInput = document.getElementById("search-input");
        const searchResults = document.getElementById("search-results");

        let timer;

        // Show popup
        function showPopup() {
            searchResults.style.display = 'block';
        }

        // Hide popup
        function hidePopup() {
            searchResults.style.display = 'none';
        }

        searchInput.addEventListener("keyup", () => {
            clearTimeout(timer);

            timer = setTimeout(async () => {
                const query = searchInput.value.trim();

                if(query === ""){
                    searchResults.innerHTML = "";
                    hidePopup();
                    return;
                }

                try {
                    const response = await fetch(`/api/search?q=${query}`);
                    const result = await response.json();

                    searchResults.innerHTML = "";

                    // USERS
                    result.data.users.forEach(user => {
                        const userDiv = document.createElement('div');
                        userDiv.className = 'search-user';
                        userDiv.innerHTML = `<img src="${user.profilePic}" width="30"><span>${user.username}</span>`;
                        
                        // Navigate to user page when clicked
                        userDiv.addEventListener('click', () => {
                            window.location.href = `/SearchUser.html?userId=${user._id}`;
                        });

                        searchResults.appendChild(userDiv);
                    });

                    // POSTS
                    result.data.posts.forEach(post => {
                        const postDiv = document.createElement('div');
                        postDiv.className = 'search-post';
                        postDiv.innerHTML = `<b>${post.user.username}</b><p>${post.content}</p>`;
                        
                        // Navigate to post page when clicked
                        postDiv.addEventListener('click', () => {
                            window.location.href = `/PostPage.html?postId=${post._id}`;
                        });

                        searchResults.appendChild(postDiv);
                    });

                    showPopup();

                } catch(error){
                    console.error("Search error:", error);
                }
            }, 300);
        });

        // Hide popup when clicking outside
        document.addEventListener('click', (e) => {
            if(!searchResults.contains(e.target) && e.target !== searchInput){
                hidePopup();
            }
        });
    </script>
